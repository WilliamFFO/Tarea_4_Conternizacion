2. Descripción

Se descargó el dataset público Pokemon with stats (800 filas, 13 columnas, CSV), se cargó completo en una base de datos serverless (Cloudflare D1) y se desarrolló una aplicación web serverless que permite visualizar los datos en tarjetas, buscarlos por ID (llave primaria) o por nombre (columna única), filtrarlos por tipo, generación y legendarios, y ver una gráfica con el promedio de puntos por tipo.

LógicaCloudflare Workers (función serverless)src/worker.jsSe ejecuta solo cuando llega una petición; escala sola; el navegador nunca habla directo con la base de datos

La columna original «#» se repite (solo 721 valores distintos en 800 filas, por las mega-evoluciones), por eso la tabla usa un id propio como llave primaria y name (única) como columna secundaria de búsqueda.

3. Script de la base de datos

Estructura de la tabla db/schema.sql

-- =====================================================================
-- Estructura de la base de datos serverless (Cloudflare D1, motor SQLite)
-- Dataset: Pokemon with stats (800 filas, 13 columnas originales)
--
-- Notas de diseno:
--   * "id" es la llave primaria propia (1..800). La columna original "#"
--     NO es unica (las mega-evoluciones repiten el numero), por eso se
--     guarda aparte como "dex_number".
--   * "name" es UNICA: sirve como columna secundaria de busqueda.
--   * "type2" admite NULL (muchos pokemon tienen un solo tipo).
--   * "legendary" es booleano guardado como 0/1 (SQLite no tiene BOOLEAN).
-- =====================================================================

DROP TABLE IF EXISTS pokemon;

CREATE TABLE pokemon (
  id          INTEGER PRIMARY KEY,           -- llave primaria (1..800)
  dex_number  INTEGER NOT NULL,              -- columna original "#"
  name        TEXT    NOT NULL UNIQUE,       -- nombre unico
  type1       TEXT    NOT NULL,              -- tipo principal
  type2       TEXT,                          -- tipo secundario (puede ser NULL)
  total       INTEGER NOT NULL,              -- suma de las 6 estadisticas
  hp          INTEGER NOT NULL,
  attack      INTEGER NOT NULL,
  defense     INTEGER NOT NULL,
  sp_atk      INTEGER NOT NULL,              -- columna original "Sp. Atk"
  sp_def      INTEGER NOT NULL,              -- columna original "Sp. Def"
  speed       INTEGER NOT NULL,
  generation  INTEGER NOT NULL,              -- generacion 1..6
  legendary   INTEGER NOT NULL DEFAULT 0     -- 0 = no, 1 = si
);

-- Indices para acelerar las busquedas y filtros de la aplicacion
CREATE INDEX idx_pokemon_type1      ON pokemon (type1);
CREATE INDEX idx_pokemon_type2      ON pokemon (type2);
CREATE INDEX idx_pokemon_generation ON pokemon (generation);

// Uso: node scripts/csv_to_sql.mjs
import { readFileSync, writeFileSync } from 'node:fs';
import { fileURLToPath } from 'node:url';
import { dirname, join } from 'node:path';

const root = join(dirname(fileURLToPath(import.meta.url)), '..');
const csv = readFileSync(join(root, 'data', 'pokemon.csv'), 'utf8').replace(/^﻿/, '');

// Lector de CSV minimo que respeta comillas
function parseCsv(text) {
  const rows = [];
  let row = [], field = '', quoted = false;
  for (let i = 0; i < text.length; i++) {
    const ch = text[i];
    if (quoted) {
      if (ch === '"' && text[i + 1] === '"') { field += '"'; i++; }
      else if (ch === '"') quoted = false;
      else field += ch;
    } else if (ch === '"') quoted = true;
    else if (ch === ',') { row.push(field); field = ''; }
    else if (ch === '\n') { row.push(field.replace(/\r$/, '')); rows.push(row); row = []; field = ''; }
    else field += ch;
  }
  if (field !== '' || row.length) { row.push(field.replace(/\r$/, '')); rows.push(row); }
  return rows.filter((r) => r.length > 1);
}

const [header, ...data] = parseCsv(csv);
if (header.length !== 13) throw new Error(`Se esperaban 13 columnas y hay ${header.length}`);

const q = (s) => `'${String(s).replace(/'/g, "''")}'`;
const int = (s, col) => {
  const n = Number(s);
  if (!Number.isInteger(n)) throw new Error(`Valor no entero en ${col}: "${s}"`);
  return n;
};

const names = new Set();
const tuples = data.map((r, i) => {
  const [dex, name, t1, t2, total, hp, atk, def, spa, spd, spe, gen, leg] = r;
  if (names.has(name)) throw new Error(`Nombre repetido: ${name}`);
  names.add(name);
  return '(' + [
    i + 1, int(dex, '#'), q(name), q(t1), t2 === '' ? 'NULL' : q(t2),
    int(total, 'Total'), int(hp, 'HP'), int(atk, 'Attack'), int(def, 'Defense'),
    int(spa, 'Sp. Atk'), int(spd, 'Sp. Def'), int(spe, 'Speed'), int(gen, 'Generation'),
    leg.trim().toLowerCase() === 'true' ? 1 : 0,
  ].join(', ') + ')';
});

const cols = 'id, dex_number, name, type1, type2, total, hp, attack, defense, sp_atk, sp_def, speed, generation, legendary';
const BATCH = 25;
let sql = `-- Generado por scripts/csv_to_sql.mjs a partir de data/pokemon.csv\n-- Filas: ${tuples.length}\n\n`;
for (let i = 0; i < tuples.length; i += BATCH) {
  sql += `INSERT INTO pokemon (${cols}) VALUES\n  ${tuples.slice(i, i + BATCH).join(',\n  ')};\n\n`;
}
writeFileSync(join(root, 'db', 'seed.sql'), sql, 'utf8');
console.log(`OK: ${tuples.length} filas -> db/seed.sql`);

Función serverless (API) src/worker.js

// Funcion serverless (Cloudflare Worker). No hay servidor que administrar:
// Cloudflare la ejecuta solo cuando llega una peticion a /api/* y la apaga despues.
// La base de datos (D1) llega a traves del "binding" env.DB definido en wrangler.toml.

const COLUMNS =
  'id, dex_number, name, type1, type2, total, hp, attack, defense, sp_atk, sp_def, speed, generation, legendary';

const json = (data, status = 200, headers = {}) =>
  new Response(JSON.stringify(data), {
    status,
    headers: { 'content-type': 'application/json; charset=utf-8', ...headers },
  });

// Busqueda con filtros y paginacion.
//   q: si es un numero se busca por ID (llave primaria); si es texto, por nombre (columna unica)
async function listPokemon(url, env) {
  const q = (url.searchParams.get('q') || '').trim().slice(0, 60);
  const type = (url.searchParams.get('type') || '').trim().slice(0, 20);
  const gen = Number.parseInt(url.searchParams.get('gen'), 10);
  const legendary = url.searchParams.get('legendary') === '1';
  const page = Math.max(1, Number.parseInt(url.searchParams.get('page'), 10) || 1);
  const limit = Math.min(48, Math.max(1, Number.parseInt(url.searchParams.get('limit'), 10) || 24));

  const where = [];
  const params = [];
  if (q) {
    if (/^\d+$/.test(q)) {
      where.push('id = ?');
      params.push(Number(q));
    } else {
      where.push("name LIKE ? ESCAPE '\\'");
      params.push(`%${q.replace(/[\\%_]/g, '\\$&')}%`);
    }
  }
  if (type) {
    where.push('(type1 = ? OR type2 = ?)');
    params.push(type, type);
  }
  if (Number.isInteger(gen)) {
    where.push('generation = ?');
    params.push(gen);
  }
  if (legendary) where.push('legendary = 1');
  const clause = where.length ? `WHERE ${where.join(' AND ')}` : '';

  // Consultas parametrizadas (bind): evitan inyeccion SQL
  const [count, rows] = await env.DB.batch([
    env.DB.prepare(`SELECT COUNT(*) AS n FROM pokemon ${clause}`).bind(...params),
    env.DB.prepare(`SELECT ${COLUMNS} FROM pokemon ${clause} ORDER BY id LIMIT ? OFFSET ?`).bind(
      ...params,
      limit,
      (page - 1) * limit,
    ),
  ]);
  const total = count.results[0].n;
  return json({ total, page, pages: Math.max(1, Math.ceil(total / limit)), limit, results: rows.results });
}

async function getPokemon(id, env) {
  const row = await env.DB.prepare(`SELECT ${COLUMNS} FROM pokemon WHERE id = ?`).bind(id).first();
  return row ? json(row) : json({ error: `No existe un Pokémon con ID ${id}` }, 404);
}

// Datos agregados para las tarjetas resumen y la grafica
async function getStats(env) {
  const [resumen, porTipo, porGeneracion] = await env.DB.batch([
    env.DB.prepare(
      'SELECT COUNT(*) AS total, SUM(legendary) AS legendarios, ROUND(AVG(total), 1) AS promedio_total FROM pokemon',
    ),
    env.DB.prepare(
      'SELECT type1 AS tipo, COUNT(*) AS cantidad, ROUND(AVG(total), 1) AS promedio_total FROM pokemon GROUP BY type1 ORDER BY promedio_total DESC',
    ),
    env.DB.prepare('SELECT generation, COUNT(*) AS cantidad FROM pokemon GROUP BY generation ORDER BY generation'),
  ]);
  return json(
    { resumen: resumen.results[0], porTipo: porTipo.results, porGeneracion: porGeneracion.results },
    200,
    { 'cache-control': 'public, max-age=60' },
  );
}

async function getTypes(env) {
  const { results } = await env.DB.prepare(
    'SELECT type1 AS tipo FROM pokemon UNION SELECT type2 FROM pokemon WHERE type2 IS NOT NULL ORDER BY 1',
  ).all();
  return json(results.map((r) => r.tipo), 200, { 'cache-control': 'public, max-age=300' });
}

export default {
  async fetch(request, env) {
    const url = new URL(request.url);
    if (request.method !== 'GET') return json({ error: 'Método no permitido' }, 405);
    try {
      if (url.pathname === '/api/pokemon') return await listPokemon(url, env);
      const m = url.pathname.match(/^\/api\/pokemon\/(\d+)$/);
      if (m) return await getPokemon(Number(m[1]), env);
      if (url.pathname === '/api/stats') return await getStats(env);
      if (url.pathname === '/api/types') return await getTypes(env);
      return json({ error: 'Ruta no encontrada' }, 404);
    } catch (err) {
      console.error(err);
      return json({ error: 'Error interno al consultar la base de datos' }, 500);
    }
  },
};

const plain = (s) => s.toLowerCase().replace('♀', '-f').replace('♂', '-m').replace(/[.'’%]/g, '').replace(/é/g, 'e')
  .replace(/[^a-z0-9-]+/g, '-').replace(/^-+|-+$/g, '');
function spriteSlugs(rawName) {
  const name = rawName.replace(/50% Forme$/, '');
  // "CharizardMega Charizard X" -> charizard-mega-x ; "KyogrePrimal Kyogre" -> kyogre-primal
  let m = name.match(/^(.+?)(Mega|Primal) .*?(?: ([XY]))?$/);
  if (m) {
    const base = plain(m[1]);
    return [`${base}-${m[2].toLowerCase()}${m[3] ? `-${m[3].toLowerCase()}` : ''}`, base];
  }
  // "DeoxysAttack Forme" -> deoxys-attack ; "HoopaHoopa Unbound" -> hoopa-unbound
  m = name.match(/^([A-Z][a-zé'.-]*[a-zé])([A-Z].*)$/);
  if (m) {
    const base = plain(m[1]);
    const words = m[2].split(' ');
    const form = words[0] === m[1] ? words[1] : words[0];
    return [`${base}-${plain(form)}`, base];
  }
  return [plain(name)];
}
// Nombre para mostrar: "VenusaurMega Venusaur" -> "Venusaur Mega" ; "DeoxysAttack Forme" -> "Deoxys (Attack)".
// En la base de datos el nombre no se modifica (la búsqueda usa el original).
function prettyName(raw) {
  const name = raw.replace(/(\D)50% Forme$/, '$1 50%');
  let m = name.match(/^(.+?)(Mega|Primal) .*?(?: ([XY]))?$/);
  if (m) return `${m[1]} ${m[2]}${m[3] ? ` ${m[3]}` : ''}`;
  m = name.match(/^([A-Z][a-zé'.-]*[a-zé])([A-Z].*)$/);
  if (m) {
    const form = m[2].replace(`${m[1]} `, '').replace(new RegExp(` ${m[1]}$`), '').replace(/ (Forme|Cloak|Mode|Size)$/, '');
    return `${m[1]} (${form})`;
  }
  return name;
}

};

async function search(page = 1) {
  const params = new URLSearchParams({ page });
  if ($('q').value.trim()) params.set('q', $('q').value.trim());
  if ($('type').value) params.set('type', $('type').value);
  if ($('gen').value) params.set('gen', $('gen').value);
  if ($('legendary').checked) params.set('legendary', '1');

  const status = $('status');
  status.className = 'status';
  status.textContent = 'Buscando…';
  try {
    const data = await api(`/api/pokemon?${params}`);
    state = { page: data.page, pages: data.pages };
    $('cards').innerHTML = data.results.map(cardHtml).join('');
    status.textContent = data.total
      ? `${data.total} resultado${data.total === 1 ? '' : 's'}`
      : 'No se encontraron Pokémon con esos criterios.';
    $('pageinfo').textContent = `Página ${data.page} de ${data.pages}`;
    $('prev').disabled = data.page <= 1;

  } catch (err) {
    $('cards').innerHTML = '';
    status.className = 'status error';
    status.textContent = `No se pudo consultar la base de datos: ${err.message}`;
  }
}
